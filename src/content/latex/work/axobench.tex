\begin{publicationsummary}
\publicationsummarypart{Purpose}{AxoBench measures how faithfully learned neuron models reproduce detailed neuron simulations, with datasets, metrics, and an evaluation protocol. It scores any NumPy prediction function and needs only NumPy, Matplotlib, and Requests.}
\publicationsummarypart{Datasets}{Its three Hugging Face datasets share one format: inputs in 1\,ms steps on 1,278 synaptic channels (639 excitatory, 639 inhibitory), with the neuron's spikes and normalized voltage as targets. They differ in where the inputs come from, and so in the question each answers.}
\end{publicationsummary}
