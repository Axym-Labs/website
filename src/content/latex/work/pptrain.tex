\begin{publicationsummary}
\publicationsummarypart{Area}{\textbf{Task-agnostic pretraining} trains a model on synthetic or procedural tasks selected for reusable structure rather than for one downstream objective. It seeks patterns that transfer across later tasks, including compositional rules, state transitions, symbolic operations, and compression strategies.}
\publicationsummarypart{Prepretraining}{Prepretraining is one implementation: the task-agnostic phase runs before training on natural data.}
\publicationsummarypart{Library}{pptrain makes task-agnostic pretraining, on synthetic tasks chosen for structure that transfers, available as one installable workflow with six paper-derived task families and transfer evaluation.}
\end{publicationsummary}
